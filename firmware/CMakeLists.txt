cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico CACHE STRING "Only original RP2040 Pico/Pico H is supported")
if(NOT PICO_BOARD STREQUAL "pico")
    message(FATAL_ERROR "This prototype is for an original Pico / Pico H, not Pico W or Pico 2")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(moza_ps5_diagnostic C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(moza_ps5_diagnostic main.c usb_descriptors.c core/auth.c core/protocol.c core/usb_reports.c core/diagnostics.c core/output_stream.c)
target_include_directories(moza_ps5_diagnostic PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(moza_ps5_diagnostic PRIVATE -Wall -Wextra -Werror)
option(MP_FIRMWARE_V2 "Build the experimental compact G29-style PS5 firmware" OFF)
option(MP_FIRMWARE_V3 "Build the upstream-derived G29 USB profile experiment" OFF)
option(MP_OUTPUT_STREAM "Opt-in complete USB output UART stream (v3 only)" OFF)
if(MP_OUTPUT_STREAM AND NOT MP_FIRMWARE_V3)
    message(FATAL_ERROR "MP_OUTPUT_STREAM requires MP_FIRMWARE_V3")
endif()
if(MP_OUTPUT_STREAM)
    target_compile_definitions(moza_ps5_diagnostic PRIVATE MP_OUTPUT_STREAM=1)
endif()
if(MP_FIRMWARE_V2 AND MP_FIRMWARE_V3)
    message(FATAL_ERROR "Select either MP_FIRMWARE_V2 or MP_FIRMWARE_V3, not both")
endif()
if(MP_FIRMWARE_V2)
    target_compile_definitions(moza_ps5_diagnostic PRIVATE MP_FIRMWARE_V2=1)
elseif(MP_FIRMWARE_V3)
    target_compile_definitions(moza_ps5_diagnostic PRIVATE MP_FIRMWARE_V3=1)
endif()
target_link_libraries(moza_ps5_diagnostic pico_stdlib hardware_uart hardware_irq tinyusb_device tinyusb_board)
pico_enable_stdio_usb(moza_ps5_diagnostic 0)
pico_enable_stdio_uart(moza_ps5_diagnostic 0)
pico_set_program_name(moza_ps5_diagnostic "MOZA PS5 authentication diagnostic")
if(MP_OUTPUT_STREAM)
    pico_set_program_version(moza_ps5_diagnostic "0.3.1-outputs-experimental")
elseif(MP_FIRMWARE_V3)
    pico_set_program_version(moza_ps5_diagnostic "0.3.0-experimental")
elseif(MP_FIRMWARE_V2)
    pico_set_program_version(moza_ps5_diagnostic "0.2.0-experimental")
else()
    pico_set_program_version(moza_ps5_diagnostic "0.1.0")
endif()
pico_add_extra_outputs(moza_ps5_diagnostic)
